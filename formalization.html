Reference Python Verification Script — Balancing Multiplier, Sets & 5 OEIS Sequences

Self-contained standard-library Python 3 script reproducing exact terms for OEIS A382286, A401528, A381054, A382306, and A383115, verifying the sharp prime ceiling β(p) = ⌊√p⌋² and 4:1 spectral gap, and evaluating empirical convergence toward Apéry’s constant ζ(3). 

import math

def central_divisors(n: int) -> tuple[int, int]:
    s = math.isqrt(n)
    while n % s != 0:
        s -= 1
    return s, n // s

def shell_discrepancy(n: int) -> int:
    d1, d2 = central_divisors(n)
    return math.isqrt(d2) - math.isqrt(d1)

# OEIS A382286: Minimal balancing multiplier beta(n)
def a382286_beta(n: int) -> int:
    if n <= 1 or shell_discrepancy(n) == 0:
        return 1
    k = 2
    while shell_discrepancy(k * n) != 0:
        k += 1
    return k

# OEIS A381054: Minimal multiplier for unit shell discrepancy
def a381054_unit(n: int) -> int:
    k = 1
    while shell_discrepancy(k * n) != 1:
        k += 1
    return k

# OEIS A382306: Cardinality of the balanced set in shell S_k
def a382306_balanced_count(k: int) -> int:
    return sum(1 for m in range(k * k, k * k + 2 * k + 1) if shell_discrepancy(m) == 0)

# OEIS A383115: Cardinality of the unit-discrepancy set in shell S_k
def a383115_eccentric1_count(k: int) -> int:
    return sum(1 for m in range(k * k, k * k + 2 * k + 1) if shell_discrepancy(m) == 1)

# OEIS A401528: Square-multiplier set {n >= 1 : beta(n) is a square}
def a401528_square_multiplier_set(limit: int) -> list[int]:
    out = []
    for n in range(1, limit + 1):
        b = a382286_beta(n)
        if math.isqrt(b) ** 2 == b:
            out.append(n)
    return out

if __name__ == "__main__":
    print("A382286 (1..20):", [a382286_beta(n) for n in range(1, 21)])
    print("A381054 (1..20):", [a381054_unit(n) for n in range(1, 21)])
    print("A382306 (1..15):", [a382306_balanced_count(k) for k in range(1, 16)])
    print("A383115 (1..15):", [a383115_eccentric1_count(k) for k in range(1, 16)])
    print("A401528 (<= 35):", a401528_square_multiplier_set(35))
